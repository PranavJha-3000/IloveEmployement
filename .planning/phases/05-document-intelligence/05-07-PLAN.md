Contracts this plan creates:
- type FieldFlag = { edited: boolean; confirmed: boolean }
- interface ReviewState { profile: StructuredProfile; original: StructuredProfile; confidence: Record<FieldPath, number>; flags: Record<FieldPath, FieldFlag>; errors: Record<FieldPath, string>; touched: Record<FieldPath, boolean>; attemptedConfirm: boolean }
- type ReviewAction =
    | { type: "edit"; path: FieldPath; value: unknown }
    | { type: "blur"; path: FieldPath }
    | { type: "toggleConfirm"; path: FieldPath }
    | { type: "confirmAllHigh" }
    | { type: "addItem"; group: "roles" | "education" | "projects" | "links" }
    | { type: "removeItem"; group: "roles" | "education" | "projects" | "links"; index: number }
    | { type: "setSkills"; skills: NormalizedSkill[] }
    | { type: "attemptConfirm" }
- initReview(input: { result: ExtractionResult } | { saved: ConfirmedProfile }): ReviewState   // saved -> every field confirmed, confidence all 1
- reviewReducer(s: ReviewState, a: ReviewAction): ReviewState
- fieldStatus(s: ReviewState, path: FieldPath): "high" | "low" | "empty" | "edited" | "confirmed" | "invalid"
- summaryCounts(s): { needsLook: number; confirmed: number; total: number; invalid: number }
- ProfileReview props: { initial: { result: ExtractionResult } | { saved: ConfirmedProfile }; onConfirm(profile: StructuredProfile): Promise<void>; onDiscard(): void; confirming: boolean; headerExtra?: React.ReactNode }
</interfaces>
